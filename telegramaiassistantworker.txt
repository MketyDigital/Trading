      }

    } catch (err) {

function logBg(ctx, promise) {
  if (ctx && typeof ctx.waitUntil === 'function') {
    ctx.waitUntil(promise);
  } else {
    promise.catch(() => {});
  }
}

function cleanPlainText(text) {
  if (!text) return "";
  return text
    .replace(/\*\*/g, '')
    .replace(/\*/g, '')
    .replace(/^#{1,6}\s+/gm, '')
    .replace(/^[\s-]*[-*+]\s+/gm, '')
    .replace(/```[\s\S]*?```/g, '')
    .replace(/`([^`]+)`/g, '$1')
    .replace(/\[([^\]]+)\]\([^)]+\)/g, '$1')
    .trim();
}

function getDateTimeIntelligence() {
  const now = new Date();
  const nigerianTimeStr = new Intl.DateTimeFormat("en-US", {
    timeZone: "Africa/Lagos",
    weekday: "long",
    year: "numeric",
    month: "long",
    day: "numeric",
    hour: "2-digit",
    minute: "2-digit",
    hour12: true
  }).format(now);

  const year = now.getFullYear();
  const month = now.getMonth();
  const dayOfMonth = now.getDate();

  const lastDayOfMonth = new Date(year, month + 1, 0).getDate();
  const daysLeft = lastDayOfMonth - dayOfMonth;
  const isLastWeekOfMonth = daysLeft < 7;

  const getSecondMonday = (y, m) => {
    let count = 0;
    for (let d = 1; d <= 31; d++) {
      const dt = new Date(y, m, d);
      if (dt.getMonth() !== m) break;
      if (dt.getDay() === 1) {
        count++;
        if (count === 2) return dt;
      }
    }
    return null;
  };

  const thisMonthSecondMonday = getSecondMonday(year, month);
  const nextMonthSecondMonday = getSecondMonday(year, month + 1);

  const formatDate = (dt) => {
    if (!dt) return "N/A";
    return new Intl.DateTimeFormat("en-US", { weekday: "long", month: "long", day: "numeric", year: "numeric" }).format(dt);
  };

  return `
[REAL-TIME CALENDAR & TIME INTELLIGENCE - NIGERIA / WAT TIME]
Current Real-World Date & Time: ${nigerianTimeStr}
Day of Month: ${dayOfMonth} of ${lastDayOfMonth}
Is Today in the Last Week of the Month?: ${isLastWeekOfMonth ? "YES (Free 3-Day Foundational Class Week!)" : "NO"}
This Month's Second Monday: ${formatDate(thisMonthSecondMonday)}
Next Month's Second Monday (For N50,000 Advanced Class Start): ${formatDate(nextMonthSecondMonday)}

CLASS SCHEDULE RULES (BE SMART AND HUMAN):
- Standard Live Class Time: All live classes always start at 9:00 PM (21:00 WAT).
- Free 3-Day Foundational Class: Held every last week of the month at 9:00 PM.
  * ${isLastWeekOfMonth ? "Since today is in the last week of the month, tell the customer naturally that the Free 3-Day Foundational Class is holding / starting tonight at 9:00 PM!" : "Tell the customer that registration is ongoing for this month's Free 3-Day Foundational Class coming up during the last week of the month at 9:00 PM."}
- N50,000 Advanced Group Class: Always starts on the second Monday of the next month at 9:00 PM (${formatDate(nextMonthSecondMonday)} at 9:00 PM). Tell the customer the exact starting date naturally!
- N500,000 Private 1-on-1 Class: Starts immediately upon payment (2 months intensive with lifetime mentorship).
`;
}

async function resolveBotToken(env, urlObj) {
  const tokenFromUrl = urlObj.searchParams.get("token") || urlObj.searchParams.get("bot_token");
  if (tokenFromUrl && tokenFromUrl.trim().length > 10) {
    return tokenFromUrl.trim();
  }

  if (env.SUPABASE_URL && env.SUPABASE_ANON_KEY) {
    try {
      const botIdParam = urlObj.searchParams.get("bot_id");
      const baseUrl = env.SUPABASE_URL.replace(/\/+$/, '');
      let dbUrl = `${baseUrl}/rest/v1/telegram_bots?select=token&is_active=eq.true&order=created_at.desc&limit=1`;
      if (botIdParam && botIdParam !== "global") {
        dbUrl = `${baseUrl}/rest/v1/telegram_bots?select=token&id=eq.${botIdParam}&is_active=eq.true&limit=1`;
      }
      const res = await fetch(dbUrl, {
        headers: {
          "apikey": env.SUPABASE_ANON_KEY,
          "Authorization": `Bearer ${env.SUPABASE_ANON_KEY}`,
          "Content-Type": "application/json"
        }
      });
      if (res.ok) {
        const rows = await res.json();
        if (rows && rows.length > 0 && rows[0].token) {
          return rows[0].token.trim();
        }
      }
    } catch (e) {
      console.warn("Could not query telegram_bots table:", e);
    }
  }

  if (env.TELEGRAM_BOT_TOKEN && env.TELEGRAM_BOT_TOKEN.trim().length > 10) {
    return env.TELEGRAM_BOT_TOKEN.trim();
  }

  return null;
}

async function fetchSupabaseSettings(env) {
  if (!env.SUPABASE_URL || !env.SUPABASE_ANON_KEY) return null;
  const baseUrl = env.SUPABASE_URL.replace(/\/+$/, '');
  const res = await fetch(`${baseUrl}/rest/v1/assistant_settings?select=*&order=updated_at.desc&limit=1`, {
    headers: {
      "apikey": env.SUPABASE_ANON_KEY,
      "Authorization": `Bearer ${env.SUPABASE_ANON_KEY}`,
      "Content-Type": "application/json"
    }
  });
  if (!res.ok) return null;
  const data = await res.json();
  return data.length > 0 ? data[0] : null;
}

async function fetchKnowledgeBaseContext(env, botScope) {
  if (!env.SUPABASE_URL || !env.SUPABASE_ANON_KEY) return "No active knowledge base connection.";
  const baseUrl = env.SUPABASE_URL.replace(/\/+$/, '');
  const filter = botScope && botScope !== "global"
    ? `or=(bot_scope.eq.global,bot_scope.eq.${botScope},bot_id.eq.global,bot_id.eq.${botScope})`
    : `or=(bot_scope.eq.global,bot_id.eq.global)`;

  const url = `${baseUrl}/rest/v1/knowledge_base?select=name,title,content&status=eq.active&${filter}&limit=25`;
  
  const res = await fetch(url, {
    headers: {
      "apikey": env.SUPABASE_ANON_KEY,
      "Authorization": `Bearer ${env.SUPABASE_ANON_KEY}`,
      "Content-Type": "application/json"
    }
  });
  if (!res.ok) return "No active knowledge base files found.";
  const items = await res.json();
  if (!items || items.length === 0) return "No active knowledge base files found.";
  return items.map(item => `[Doc: ${item.title || item.name || 'Knowledge'}]\n${item.content || ""}`).join("\n\n");
}

async function saveChatMessage(env, chatId, senderType, senderName, text) {
  if (!env.SUPABASE_URL || !env.SUPABASE_ANON_KEY || !text) return;
  const baseUrl = env.SUPABASE_URL.replace(/\/+$/, '');
  try {
    await fetch(`${baseUrl}/rest/v1/chat_messages`, {
      method: "POST",
      headers: {
        "apikey": env.SUPABASE_ANON_KEY,
        "Authorization": `Bearer ${env.SUPABASE_ANON_KEY}`,
        "Content-Type": "application/json",
        "Prefer": "return=minimal"
      },
      body: JSON.stringify({
        chat_id: String(chatId),
        sender_type: senderType,
        sender_name: senderName || 'User',
        message_text: text,
        created_at: new Date().toISOString()
      })
    });
  } catch (e) {
    console.warn("Could not save chat message to Supabase:", e);
  }
}

      headers: {
        "apikey": env.SUPABASE_ANON_KEY,
        "Authorization": `Bearer ${env.SUPABASE_ANON_KEY}`,
        "Content-Type": "application/json"
      }
    });
    if (res.ok) {
      const rows = await res.json();
      if (rows && rows.length > 0) {
        // Reverse so history is in chronological order (oldest to newest)
        const sortedRows = rows.reverse();
        const historyText = sortedRows.map(r => `[${r.sender_name || r.sender_type}]: ${r.message_text}`).join("\n");

async function checkChatPauseInSupabase(env, chatId) {
  if (!env.SUPABASE_URL || !env.SUPABASE_ANON_KEY) return false;
  const baseUrl = env.SUPABASE_URL.replace(/\/+$/, '');
  try {
    const res = await fetch(`${baseUrl}/rest/v1/chat_overrides?chat_id=eq.${chatId}&select=is_paused,paused_until&limit=1`, {
      headers: {
        "apikey": env.SUPABASE_ANON_KEY,
        "Authorization": `Bearer ${env.SUPABASE_ANON_KEY}`,
        "Content-Type": "application/json"
      }
    });
    if (res.ok) {
      const rows = await res.json();
      if (rows && rows.length > 0) {
        const row = rows[0];
        // If there is a paused_until timestamp, evaluate expiry
        if (row.paused_until) {
          const pauseEndTime = new Date(row.paused_until).getTime();
          if (pauseEndTime > Date.now()) {
            return true; // Still within active auto-pause window
          } else {
            return false; // Pause window expired
          }
        }
        if (row.is_paused === true) return true;
      }
    }
  } catch (e) {
    console.warn("Could not check chat pause in Supabase:", e);
  }
  return false;
}

async function generateAIResponse(env, settings, systemPrompt, userText, ctx) {
  let preferredModel = "gemini-3.5-flash";
  
  if (settings.custom_model_string && settings.custom_model_string.trim()) {
    preferredModel = settings.custom_model_string.trim();
  } else if (settings.selected_engine) {
    const engine = String(settings.selected_engine).toLowerCase();
    if (engine.includes("3.6")) {
      preferredModel = engine.includes("pro") ? "gemini-3.5-pro" : "gemini-3.6-flash";
    } else if (engine.includes("3.5")) {
      preferredModel = engine.includes("pro") ? "gemini-3.5-pro" : "gemini-3.5-flash";
    } else if (engine.includes("pro")) {
      preferredModel = "gemini-3.5-pro";
    } else if (engine.includes("4o")) {
      preferredModel = engine.includes("mini") ? "gpt-4o-mini" : "gpt-4o";
    } else {
      preferredModel = "gemini-3.5-flash";
    }
  }

  let keysToTry = [];
  const rawDbKeys = [
    settings.gemini_api_keys,
    settings.gemini_api_key,
    settings.custom_api_keys,
    settings.openai_api_keys,
    settings.claude_api_keys
  ].filter(Boolean).join(",");

  if (rawDbKeys && rawDbKeys.trim().length > 5) {
    const extracted = rawDbKeys.split(/[\n,;]+/).map(k => k.trim().replace(/^["']|["']$/g, '')).filter(k => k.length > 5);
    keysToTry.push(...extracted);
  }

  if (env.GEMINI_API_KEY && env.GEMINI_API_KEY.trim().length > 5) {
    keysToTry.push(env.GEMINI_API_KEY.trim());
  }

  keysToTry = Array.from(new Set(keysToTry));
  
  // Shuffle keys to distribute load and avoid immediate rate limits
  for (let i = keysToTry.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [keysToTry[i], keysToTry[j]] = [keysToTry[j], keysToTry[i]];
  }

  if (keysToTry.length === 0) {
    logBg(ctx, logToSupabase(env, "error", "[SYSTEM FAILOVER / ADMIN ACTION REQUIRED] Missing AI API Key. Please add your Gemini API Key in your Portal under AI Engine Settings or Worker env."));
    return "Please hold on a bit let me get back to you on that.";
  }

  const modelsToTry = [
    preferredModel, 
    "gemini-3.5-flash", 
    "gemini-3.6-flash", 
    "gemini-3.5-pro"
  ];
  const uniqueModels = Array.from(new Set(modelsToTry));

  const geminiPayload = {
    contents: [
      {
        role: "user",
        parts: [{ text: `System Role & Directives:\n${systemPrompt}\n\nUser Question:\n"${userText}"` }]
      }
    ],
    generationConfig: {
      temperature: settings.temperature || 0.4,
      maxOutputTokens: settings.max_tokens || 1500
    }
  };

  let lastErrorDetail = "";

  // Outer loop: Try each API key (Load balancing across available API Keys)
  for (const apiKey of keysToTry) {
    if (apiKey.startsWith("sk-")) {
      const openAiModel = preferredModel.includes("mini") ? "gpt-4o-mini" : "gpt-4o";
      try {
        const oaRes = await fetch("https://api.openai.com/v1/chat/completions", {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${apiKey}`
          },
          body: JSON.stringify({
            model: openAiModel,
            messages: [
              { role: "system", content: systemPrompt },
              { role: "user", content: userText }
            ],
            temperature: settings.temperature || 0.4,
            max_tokens: settings.max_tokens || 1500
          })
        });

        if (oaRes.ok) {
          const oaData = await oaRes.json();
          if (oaData.choices && oaData.choices[0] && oaData.choices[0].message) {
            return oaData.choices[0].message.content;
          }
        } else {
          lastErrorDetail = `OpenAI API Error (${oaRes.status}): ${await oaRes.text()}`;
          logBg(ctx, logToSupabase(env, "error", `[SYSTEM FAILOVER] ${lastErrorDetail.substring(0, 200)}`));
        }
      } catch (e) {
        lastErrorDetail = `OpenAI Exception: ${e.message}`;
      }
      continue;
    }

    // Try Gemini Models for this API Key
    for (const model of uniqueModels) {
      if (model.includes("gpt")) continue;
      
      const url = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`;
      try {
        const res = await fetch(url, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(geminiPayload)
        });

        if (res.ok) {
          const responseData = await res.json();
          if (responseData.candidates && responseData.candidates[0] && responseData.candidates[0].content && responseData.candidates[0].content.parts) {
            const reply = responseData.candidates[0].content.parts.map(p => p.text).join("\n");
            if (reply && reply.trim()) return reply.trim();
          }
        } else {
          const errText = await res.text();
          lastErrorDetail = `Gemini API (HTTP ${res.status} for ${model}): ${errText}`;
          if (res.status === 429) {
            logBg(ctx, logToSupabase(env, "error", `[RATE LIMIT] API Key ${apiKey.substring(0, 8)}... hit HTTP 429 on ${model}. Rotating to next API Key...`));
            // Switch to next API Key immediately!
            break;
          } else {
            logBg(ctx, logToSupabase(env, "error", `[SYSTEM FAILOVER] ${lastErrorDetail.substring(0, 180)}`));
          }
        }
      } catch (err) {
        lastErrorDetail = `Gemini Request Error (${model}): ${err.message}`;
        logBg(ctx, logToSupabase(env, "error", `[SYSTEM FAILOVER] ${lastErrorDetail}`));
      }
    }
  }

  logBg(ctx, logToSupabase(env, "error", `[SYSTEM FAILOVER / ADMIN ACTION REQUIRED] All AI key & model attempts failed. Last error: ${lastErrorDetail.substring(0, 180)}`));
  return "Please hold on a bit let me get back to you on that.";
}

async function sendTelegramMessage(env, botToken, chatId, text, replyToMessageId, businessConnectionId, ctx) {
  if (!botToken) {
    if (ctx) logBg(ctx, logToSupabase(env, "error", `Cannot send Telegram message: Bot Token is missing or not configured.`));
    return false;
  }
  const url = `https://api.telegram.org/bot${botToken}/sendMessage`;
  
  const payload = {
    chat_id: chatId,
    text: text
  };
  if (replyToMessageId) payload.reply_to_message_id = replyToMessageId;
  if (businessConnectionId) payload.business_connection_id = businessConnectionId;

  try {
    const res = await fetch(url, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload)
    });

    const data = await res.json();
    if (res.ok && data.ok) {
      if (ctx) logBg(ctx, logToSupabase(env, "success", `AI response delivered to Telegram ChatID: ${chatId}`));
      return true;
    } else {
      await logToSupabase(env, "error", `Telegram sendMessage error (HTTP ${res.status}): ${data.description || JSON.stringify(data)}`);
      return false;
    }
  } catch (err) {
    if (ctx) logBg(ctx, logToSupabase(env, "error", `Telegram sendMessage Exception: ${err.message}`));
    return false;
  }
}

  }

  if (path === '/api/sc/trading-accounts') {
    if (method === 'GET') {
      const dbAccounts = await fetchScTable(env, 'sc_trading_accounts');
      return workerJsonResponse({ ok: true, accounts: dbAccounts.length ? dbAccounts : globalSignalEngine.memoryStore.trading_accounts });
    }
    if (method === 'POST') {
      const body = await request.json().catch(() => ({}));
      if(!body.id) body.id = 'acc_' + Math.floor(Math.random()*10000);
      body.enabled = true;
      const idx = globalSignalEngine.memoryStore.trading_accounts.findIndex(a => a.id === body.id);
      if (idx >= 0) globalSignalEngine.memoryStore.trading_accounts[idx] = body;
      else globalSignalEngine.memoryStore.trading_accounts.unshift(body);
      const saved = await insertScTable(env, 'sc_trading_accounts', body);
      return workerJsonResponse({ ok: true, account: saved || body });
    }
  }

  if (path.startsWith('/api/sc/trading-accounts/') && method === 'DELETE') {
    const id = path.split('/').pop();
    globalSignalEngine.memoryStore.trading_accounts = globalSignalEngine.memoryStore.trading_accounts.filter(a => a.id !== id);
    await deleteScTable(env, 'sc_trading_accounts', id);
    return workerJsonResponse({ ok: true, deleted: id });
  }

  if (path === '/api/sc/settings') {
    if (method === 'GET') {
      const dbSettings = await fetchScTable(env, 'sc_settings');
      const settingsObj = {};
      if (dbSettings && Array.isArray(dbSettings)) {
        dbSettings.forEach(s => { settingsObj[s.key] = s.value; });
      }
      return workerJsonResponse({
        ok: true,
        settings: Object.keys(settingsObj).length > 0 ? settingsObj : globalSignalEngine.memoryStore.settings
      });
    }
    if (method === 'POST') {
      const body = await request.json().catch(() => ({}));
      if (body.key && body.value !== undefined) {
        globalSignalEngine.memoryStore.settings[body.key] = body.value;
        await insertScTable(env, 'sc_settings', { key: body.key, value: body.value, updated_at: new Date().toISOString() });
        return workerJsonResponse({ ok: true, saved: { key: body.key, value: body.value } });
      }
      if (body.settings && typeof body.settings === 'object') {
        Object.assign(globalSignalEngine.memoryStore.settings, body.settings);
        for (const [k, v] of Object.entries(body.settings)) {
          await insertScTable(env, 'sc_settings', { key: k, value: v, updated_at: new Date().toISOString() });
        }
        return workerJsonResponse({ ok: true, settings: globalSignalEngine.memoryStore.settings });
      }
      return workerJsonResponse({ ok: false, error: 'Invalid settings payload' }, 400);
    }
  }

  if (path === '/api/sc/signals' && method === 'GET') {
    const dbSignals = await fetchScTable(env, 'sc_signals');
    return workerJsonResponse({
      ok: true,
      signals: dbSignals.length ? dbSignals : globalSignalEngine.memoryStore.signals
    });
  }

  if (path === '/api/sc/audit-logs' && method === 'GET') {
    return workerJsonResponse({
      ok: true,
      audit_logs: globalSignalEngine.memoryStore.audit_logs
    });
  }

  if (path === '/api/sc/mt5/pending-commands' && method === 'GET') {
    const accId = urlObj.searchParams.get('account_id') || 'all';
    let filter = '?status=eq.PENDING';
    if (accId !== 'all') filter += `&account_id=eq.${accId}`;
    
    try {
      const res = await fetch(`${env.SUPABASE_URL}/rest/v1/sc_trading_commands${filter}`, {
        headers: {
          "apikey": env.SUPABASE_ANON_KEY,
          "Authorization": `Bearer ${env.SUPABASE_ANON_KEY}`,
          "Content-Type": "application/json"
        }
      });
      const pending = await res.json().catch(() => ([]));
      return workerJsonResponse({ ok: true, pending_commands: pending });
    } catch (err) {
      return workerJsonResponse({ ok: false, error: err.message });
    }
  }

  if (path === '/api/sc/mt5/execution-report' && method === 'POST') {
    const body = await request.json().catch(() => ({}));
    if (body.command_id && body.status) {
      try {
        const res = await fetch(`${env.SUPABASE_URL}/rest/v1/sc_trading_commands?command_id=eq.${body.command_id}`, {
          method: "PATCH",
          headers: {
            "apikey": env.SUPABASE_ANON_KEY,
            "Authorization": `Bearer ${env.SUPABASE_ANON_KEY}`,
            "Content-Type": "application/json"
          },
          body: JSON.stringify({ status: body.status, updated_at: new Date().toISOString() })
        });
        return workerJsonResponse({ ok: res.ok, status: body.status });
      } catch (err) {
        return workerJsonResponse({ ok: false, error: err.message });
      }
    }
    return workerJsonResponse({ ok: false, error: 'missing fields' });
  }

  return workerJsonResponse({ ok: false, error: 'Endpoint not found' }, 404);
}
